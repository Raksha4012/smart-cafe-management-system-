/* ============= NOTIFICATION CLASS ============= */
class Notification {
public:
    string orderID;
    string username;
    string message;
    string timestamp;
    bool isRead;

    Notification(string oID, string user, string msg, string time)
        : orderID(oID), username(user), message(msg), timestamp(time), isRead(false) {}
};

/* ============= ORDER ITEM WITH COMMENTS ============= */
class OrderItem {
public:
    string itemName;
    double price;
    int quantity;
    string comment;
    int prepTime;

    OrderItem(string name, double p, int q, int prep, string c = "")
        : itemName(name), price(p), quantity(q), prepTime(prep), comment(c) {}

    double total() { return price * quantity; }
    int prepTotal() { return prepTime * quantity; }
};

/* ============= UTILITIES ============= */
string generateOrderID() {
    srand(time(0));
    stringstream ss;
    ss << "CAF" << rand() % 10000;
    return ss.str();
}

bool userExists(const string& username) {
    return allUsers.find(username) != allUsers.end();
}

bool isValidEmail(const string& email) {
    return email.find('@') != string::npos && email.find('.') != string::npos;
}

bool isValidPhone(const string& phone) {
    return phone.length() == 10 && all_of(phone.begin(), phone.end(), ::isdigit);
}

string getCurrentDate() {
    time_t now = time(0);
    tm* timeinfo = localtime(&now);
    stringstream ss;
    ss << (timeinfo->tm_year + 1900) << "-" << setfill('0') << setw(2) 
       << (timeinfo->tm_mon + 1) << "-" << setw(2) << timeinfo->tm_mday;
    return ss.str();
}

string getCurrentTime() {
    time_t now = time(0);
    tm* timeinfo = localtime(&now);
    stringstream ss;
    ss << setfill('0') << setw(2) << timeinfo->tm_hour << ":" 
       << setw(2) << timeinfo->tm_min << ":" << setw(2) << timeinfo->tm_sec;
    return ss.str();
}

}

/* ============= RANDOM MEAL ============= */
void randomMeal() {
    double budget;
    cout << "\nEnter your budget: ₹";
    cin >> budget;

    if (budget <= 0) {
        cout << "❌ Invalid budget!\n";
        return;
    }

    srand(time(0));
    double total = 0;
    vector<MenuItem> randomItems;

    while (true) {
        int r = rand() % menu.size();
        if (total + menu[r].price > budget)
            break;
        randomItems.push_back(menu[r]);
        total += menu[r].price;
    }

    if (randomItems.empty()) {
        cout << "❌ No items available within your budget!\n";
        return;
    }

    cout << "\n✓ Suggested items within your budget:\n";
    for (int i = 0; i < randomItems.size(); i++) {
        cout << (i + 1) << ". " << randomItems[i].name << " ₹" << randomItems[i].price << endl;
    }
    cout << "Total: ₹" << total << endl;

    char ch;
    cout << "Add all to cart? (y/n): ";
    cin >> ch;
    if (ch == 'y' || ch == 'Y') {
        for (auto& item : randomItems) {
            addToCart(item, 1);
        }
    }
}

/* ============= COMBO GENERATOR ============= */
void comboGenerator() {
    srand(time(0));
    vector<MenuItem> starters, mains, drinks;

    for (auto& m : menu) {
        if (m.category == "Starter")
            starters.push_back(m);
        if (m.category == "Main")
            mains.push_back(m);
        if (m.category == "Drink")
            drinks.push_back(m);
    }

    if (starters.empty() || mains.empty() || drinks.empty()) {
        cout << "❌ Cannot generate combo - insufficient items in categories\n";
        return;
    }

    MenuItem s = starters[rand() % starters.size()];
    MenuItem m = mains[rand() % mains.size()];
    MenuItem d = drinks[rand() % drinks.size()];

    double comboPrice = (s.price + m.price + d.price) * 0.85;

    cout << "\n========== TODAY'S COMBO (15% OFF) ==========\n";
    cout << s.name << " (₹" << s.price << ")\n";
    cout << " + " << m.name << " (₹" << m.price << ")\n";
    cout << " + " << d.name << " (₹" << d.price << ")\n";
    cout << "Original Price: ₹" << (s.price + m.price + d.price) << endl;
    cout << "Combo Price: ₹" << comboPrice << endl;

    int qty = 0;
    cout << "Enter combo quantity (0 to skip): ";
    cin >> qty;

    if (qty > 0) {
        string comment;
        cout << "Any special comments? (Press Enter to skip): ";
        cin.ignore();
        getline(cin, comment);

        addToCart(s, qty, comment);
        addToCart(m, qty, comment);
        addToCart(d, qty, comment);
    }
}

/* ============= CHECKOUT & PAYMENT ============= */
void checkout() {
    if (cart.empty()) {
        cout << "\n❌ Cart is empty!\n";
        return;
    }

    if (currentUser == nullptr) {
        cout << "\n❌ Please login first!\n";
        return;
    }

    double subtotal = 0;
    int prepTotal = 0;

    for (auto& c : cart) {
        subtotal += c.total();
        prepTotal += c.prepTotal();
    }

    double tax = subtotal * 0.05;
    double discount = (subtotal > 500) ? subtotal * 0.10 : 0;
    double finalAmount = subtotal + tax - discount;

    string orderID = generateOrderID();

    cout << "\n========== RECEIPT ==========\n";
    cout << "Order ID: " << orderID << endl;
    cout << "Customer: " << currentUser->username << endl;
    cout << "________________________________\n";

    for (auto& c : cart) {
        cout << c.itemName << " x" << c.quantity << " ₹" << c.total();
        if (!c.comment.empty()) {
            cout << "\n   📝 Note: " << c.comment;
        }
        cout << endl;
    }

    cout << "________________________________\n";
    cout << "Subtotal: ₹" << subtotal << endl;
    cout << "Tax (5%): ₹" << tax << endl;

    if (discount > 0) {
        cout << "Discount (10%): -₹" << discount << endl;
    }

    cout << "Final Amount: ₹" << finalAmount << endl;
    cout << "Estimated Prep Time: ~" << prepTotal << " minutes\n";

    cout << "\n========== PAYMENT SPLIT ==========\n";
    cout << "1. Pay full amount yourself\n";
    cout << "2. Split payment with others\n";
    int splitChoice;
    cout << "Select option: ";
    cin >> splitChoice;

    if (splitChoice == 2) {
        int people;
        cout << "Split between how many people (including yourself): ";
        cin >> people;

        if (people <= 0) {
            cout << "❌ Invalid number of people!\n";
            return;
        }

        if (people > 1) {
            cout << "\n--- SPLIT BREAKDOWN ---\n";
            cout << "Total Amount: ₹" << finalAmount << endl;
            cout << "Number of people: " << people << endl;
            cout << "Each person pays: ₹" << (finalAmount / people) << endl;
        }
    } else if (splitChoice != 1) {
        cout << "❌ Invalid choice!\n";
        return;
    }

    cout << "\n========== PAYMENT METHOD ==========\n";
    cout << "1. Card\n";
    cout << "2. UPI\n";
    cout << "3. Wallet (Balance: ₹" << currentUser->walletBalance << ")\n";
    int payMethod;
    cout << "Select payment method: ";
    cin >> payMethod;

    bool paymentSuccess = false;

    if (payMethod == 1) {
        string cardNum;
        cout << "Enter Card Number (last 4 digits): ";
        cin >> cardNum;
        cout << "✓ Payment successful via Card!\n";
        paymentSuccess = true;
    } else if (payMethod == 2) {
        string upiId;
        cout << "Enter UPI ID: ";
        cin.ignore();
        getline(cin, upiId);
        cout << "✓ Payment successful via UPI!\n";
        paymentSuccess = true;
    } else if (payMethod == 3) {
        if (currentUser->walletBalance >= finalAmount) {
            currentUser->walletBalance -= finalAmount;
            cout << "✓ Payment successful via Wallet!\n";
            cout << "Remaining Wallet Balance: ₹" << currentUser->walletBalance << endl;
            paymentSuccess = true;
        } else {
            cout << "❌ Insufficient wallet balance!\n";
            return;
        }
    } else {
        cout << "❌ Invalid payment method!\n";
        return;
    }

    if (paymentSuccess) {
        // CREATE ORDER RECORD
        OrderRecord order;
        order.orderID = orderID;
        order.username = currentUser->username;
        order.email = currentUser->email;
        order.phoneNumber = currentUser->phoneNumber;
        order.items = cart;
        order.subtotal = subtotal;
        order.tax = tax;
        order.discount = discount;
        order.finalAmount = finalAmount;
        order.paymentMethod = (payMethod == 1) ? "Card" : (payMethod == 2) ? "UPI" : "Wallet";
        order.orderDate = getCurrentDate();
        order.orderTime = getCurrentTime();
        order.estimatedPrepTime = prepTotal;
        order.isCompleted = false;

        allOrders.push_back(order);
        currentUser->orderHistory.push_back(orderID);
        currentUser->totalSpent += finalAmount;

        // Update trending
        for (auto& item : cart) {
            trendingDishes[item.itemName] += item.quantity;
        }

        // Add to kitchen queue
        for (auto& item : cart) {
            KitchenPrepItem kItem(item.itemName, item.quantity, item.comment, item.prepTime, orderID);
            kItem.orderDate = order.orderDate;
            kItem.orderTime = order.orderTime;
            kitchenQueue.push_back(kItem);
        }

        // Calculate profit
        double orderProfit = 0;
        for (auto& item : cart) {
            double itemCost = item.total() * COST_PERCENTAGE;
            orderProfit += item.total() - itemCost;
        }
        orderProfit = orderProfit - (tax * COST_PERCENTAGE);
        totalProfit += orderProfit;
        totalRevenue += finalAmount;

        // Send notification to user
        sendNotification(orderID, currentUser->username, "Your order #" + orderID + " has been received! Estimated prep time: " + to_string(prepTotal) + " minutes.");

        cart.clear();
        menuDisplayed = false;
        cout << "\n========== ✓ ORDER PLACED SUCCESSFULLY! ==========\n";
        cout << "Your order has been sent to the kitchen.\n";
    }
}

/* ============= VIEW KITCHEN QUEUE ============= */
void viewKitchenQueue() {
    if (kitchenQueue.empty()) {
        cout << "\n✓ Kitchen queue is empty!\n";
        return;
    }

    cout << "\n========== KITCHEN PREPARATION QUEUE ==========\n";
    
    vector<KitchenPrepItem> sortedQueue = kitchenQueue;
    sort(sortedQueue.begin(), sortedQueue.end(),
         [](const KitchenPrepItem& a, const KitchenPrepItem& b) {
             if (a.isPrepared == b.isPrepared) {
                 return a.priority < b.priority;
             }
             return !a.isPrepared && b.isPrepared;
         });

    cout << left << setw(25) << "Item Name" << setw(8) << "Qty" << setw(10) << "Prep Time" 
         << setw(12) << "Order ID" << setw(12) << "Status" << setw(10) << "Priority\n";
    cout << string(90, '-') << endl;

    int pendingCount = 0;
    for (auto& item : sortedQueue) {
        if (!item.isPrepared) pendingCount++;

        string priorityStr;
        if (item.priority == 1) priorityStr = "⚡ HIGH";
        else if (item.priority == 2) priorityStr = "🔥 URGENT";
        else if (item.priority == 3) priorityStr = "⏱️ NORMAL";
        else if (item.priority == 4) priorityStr = "🐢 LOW";
        else priorityStr = "❄️ VERY LOW";

        string status = item.isPrepared ? "✓ Done" : "⏳ Pending";

        cout << left << setw(25) << item.itemName 
             << setw(8) << item.quantity 
             << setw(10) << (to_string(item.prepTime) + "m")
             << setw(12) << item.orderID
             << setw(12) << status
             << setw(10) << priorityStr << endl;

        if (!item.comment.empty()) {
            cout << "  📝 Note: " << item.comment << endl;
        }
    }

    cout << "\nTotal Pending Items: " << pendingCount << endl;
}

/* ============= MARK ITEM AS PREPARED & SEND NOTIFICATION ============= */
void markItemPrepared() {
    string orderID;
    cout << "\nEnter Order ID to mark as completed: ";
    cin >> orderID;

    bool found = false;
    int totalItems = 0;
    int completedItems = 0;

    // Count total items for this order
    for (auto& item : kitchenQueue) {
        if (item.orderID == orderID) {
            totalItems++;
        }
    }

    // Mark items as prepared
    for (auto& item : kitchenQueue) {
        if (item.orderID == orderID && !item.isPrepared) {
            item.isPrepared = true;
            cout << "✓ " << item.itemName << " marked as prepared!\n";
            found = true;
        }
    }

    if (!found) {
        cout << "❌ Order not found or already completed!\n";
        return;
    }

    // Check if entire order is completed
    bool allDone = true;
    for (auto& item : kitchenQueue) {
        if (item.orderID == orderID) {
            if (!item.isPrepared) {
                allDone = false;
                break;
            }
            completedItems++;
        }
    }

    if (allDone && totalItems == completedItems) {
        for (auto& order : allOrders) {
            if (order.orderID == orderID) {
                order.isCompleted = true;

                // SEND NOTIFICATION TO USER
                string message = "🎉 Your order #" + orderID + " is ready! Please come to the counter or wait for delivery.";
                sendNotification(orderID, order.username, message);

                cout << "\n✓✓ Order " << orderID << " is now READY FOR PICKUP/DELIVERY!\n";
                cout << "📢 Notification sent to customer!\n";
                break;
            }
        }
    } else {
        cout << "\n⏳ Preparing items... " << completedItems << "/" << totalItems << " items done.\n";
    }
}

    string today = getCurrentDate();

    int todayOrders = 0;

            todayOrders++;
        }
    }

    cout << "Date: " << today << endl;
    cout << "Total Orders: " << todayOrders << endl;
    cout << "Total Revenue: ₹" << todayRevenue << endl;

    cout << "\n--- CATEGORY WISE SALES ---\n";
    map<string, double> categorySales;
    map<string, int> categoryCount;

    for (auto& order : allOrders) {
        for (auto& item : order.items) {
            for (auto& menuItem : menu) {
                if (menuItem.name == item.itemName) {
                    categorySales[menuItem.category] += item.total();
                    categoryCount[menuItem.category] += item.quantity;
                    break;
                }
            }
        }
    }

    for (auto& cat : categorySales) {
        cout << cat.first << ": " << categoryCount[cat.first] << " items | ₹" << cat.second << endl;
    }

    cout << "\n--- TOP 5 CUSTOMERS ---\n";
    vector<pair<string, double>> customerSpending;
    for (auto& p : allUsers) {
        customerSpending.push_back({p.first, p.second.totalSpent});
    }
    sort(customerSpending.begin(), customerSpending.end(),
         [](const pair<string, double>& a, const pair<string, double>& b) {
             return a.second > b.second;
         });

    int count = 0;
    for (auto& cust : customerSpending) {
        if (count >= 5) break;
        cout << (count + 1) << ". " << cust.first << " - ₹" << cust.second << endl;
        count++;
    }
}
